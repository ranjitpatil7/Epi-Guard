// Test Component that consumes AuthContext
function TestComponent() {
  const { user, isAuthenticated, isLoading, login, logout } = useAuth()
  
  if (isLoading) return <div>Loading Auth...</div>
  
  return (
    <div>
      <div data-testid="auth-status">{isAuthenticated ? 'Authenticated' : 'Unauthenticated'}</div>
      <div data-testid="user-role">{user ? user.role : 'None'}</div>
      <button onClick={() => login('mock_access', 'mock_refresh', 'mock_session')}>Login</button>
      <button onClick={() => logout()}>Logout</button>
    </div>
  )
}

describe('AuthContext Integration', () => {
  beforeEach(() => {
    vi.clearAllMocks()
    localStorage.clear()
  })

  it('renders unauthenticated state initially when no token exists', async () => {
    render(<AuthProvider><TestComponent /></AuthProvider>)
    expect(screen.getByTestId('auth-status')).toHaveTextContent('Unauthenticated')
  })

  it('fetches user and sets authenticated state if token exists in localStorage', async () => {
    localStorage.setItem('access_token', 'fake_token')
    
    // Mock the /me response exactly matching backend
    vi.mocked(axios.get).mockResolvedValueOnce({
      data: {
        id: 1,
        username: 'admin_test',
        role: 'Administrator',
        region_id: null,
        permissions: ['MANAGE_USERS', 'VIEW_RISK']
      }
    })

    render(<AuthProvider><TestComponent /></AuthProvider>)
    
    expect(screen.getByText('Loading Auth...')).toBeInTheDocument()
    
    await waitFor(() => {
      expect(screen.getByTestId('auth-status')).toHaveTextContent('Authenticated')
      expect(screen.getByTestId('user-role')).toHaveTextContent('Administrator')
    })
  })

  it('handles login function and fetches user', async () => {
    vi.mocked(axios.get).mockResolvedValueOnce({
      data: { id: 2, role: 'Field Officer', permissions: [] }
    })

    render(<AuthProvider><TestComponent /></AuthProvider>)
    
    const loginButton = screen.getByText('Login')
    await userEvent.click(loginButton)

    expect(localStorage.getItem('access_token')).toBe('mock_access')
    
    await waitFor(() => {
      expect(screen.getByTestId('auth-status')).toHaveTextContent('Authenticated')
      expect(screen.getByTestId('user-role')).toHaveTextContent('Field Officer')
    })
  })

  it('handles logout function and clears state', async () => {
    // Setup initial auth state
    localStorage.setItem('access_token', 'fake')
    localStorage.setItem('session_id', 'sess1')
    
    vi.mocked(axios.get).mockResolvedValueOnce({
      data: { id: 1, role: 'Admin', permissions: [] }
    })
    
    vi.mocked(axios.post).mockResolvedValueOnce({})

    render(<AuthProvider><TestComponent /></AuthProvider>)
    
    await waitFor(() => {
      expect(screen.getByTestId('auth-status')).toHaveTextContent('Authenticated')
    })

    const logoutButton = screen.getByText('Logout')
    await userEvent.click(logoutButton)

    await waitFor(() => {
      expect(screen.getByTestId('auth-status')).toHaveTextContent('Unauthenticated')
      expect(localStorage.getItem('access_token')).toBeNull()
    })
    
    // Should call logout endpoint
    expect(axios.post).toHaveBeenCalledWith('/api/v1/auth/logout?session_id=sess1')
  })
})

    mockLastEvent = null
    
    vi.mocked(useRealtimeModule.useRealtime).mockImplementation(() => ({
      status: 'CONNECTED',
      timeline: [],
      lastEvent: mockLastEvent,
      criticalEventCount: 0,
      reconnectAttempts: 0
    } as any))

    vi.mocked(client.get).mockResolvedValue({
      data: {
        items: [
          { id: 1, name: 'ORS Packets', current_stock: 5000, minimum_threshold: 1000, unit: 'packets', is_low_stock: false },
          { id: 2, name: 'Vaccines', current_stock: 500, minimum_threshold: 1000, unit: 'vials', is_low_stock: true }
        ]
      }
    })
  })

  it('renders initial inventory from API', async () => {
    render(<InventoryDashboard />)
    
    expect(client.get).toHaveBeenCalledWith('/inventory/')
    
    await waitFor(() => {
      expect(screen.getByText('ORS Packets')).toBeInTheDocument()
      expect(screen.getByText('5000')).toBeInTheDocument()
      expect(screen.getByText('Vaccines')).toBeInTheDocument()
      expect(screen.getByText('500')).toBeInTheDocument()
    })
  })

  it('updates stock correctly when INVENTORY_UPDATED event arrives via WebSocket', async () => {
    const { rerender } = render(<InventoryDashboard />)
    
    await waitFor(() => {
      expect(screen.getByText('5000')).toBeInTheDocument()
    })

    // Simulate WS Event
    mockLastEvent = {
      event_type: 'INVENTORY_UPDATED',
      message_id: 'evt-123',
      payload: {
        item_id: 1,
        item_name: 'ORS Packets',
        quantity_deducted: 1000,
        new_stock: 4000,
        minimum_threshold: 1000,
        unit: 'packets',
        is_low_stock: false
      }
    }

    // Force re-render with the new mock value
    rerender(<InventoryDashboard />)

    // The UI should update to 4000
    await waitFor(() => {
      expect(screen.getByText('4000')).toBeInTheDocument()
      expect(screen.queryByText('5000')).not.toBeInTheDocument()
    })
  })

  it('ignores duplicate message_ids in INVENTORY_UPDATED events', async () => {
    const { rerender } = render(<InventoryDashboard />)
    
    await waitFor(() => {
      expect(screen.getByText('5000')).toBeInTheDocument()
    })

    // Event 1
    mockLastEvent = {
      event_type: 'INVENTORY_UPDATED',
      message_id: 'evt-123',
      payload: { item_id: 1, new_stock: 4000, is_low_stock: false }
    }
    rerender(<InventoryDashboard />)
    await waitFor(() => expect(screen.getByText('4000')).toBeInTheDocument())

    // If the same event ID comes in but with different stock (simulating a race condition or replay),
    // it should be ignored by the deduplication logic.
    mockLastEvent = {
      event_type: 'INVENTORY_UPDATED',
      message_id: 'evt-123', // SAME ID!
      payload: { item_id: 1, new_stock: 3000, is_low_stock: false }
    }
    rerender(<InventoryDashboard />)

    // It should STILL be 4000, not 3000
    await waitFor(() => {
      expect(screen.getByText('4000')).toBeInTheDocument()
      expect(screen.queryByText('3000')).not.toBeInTheDocument()
    })
  })
})

import { render, screen, waitFor } from '@testing-library/react'
import userEvent from '@testing-library/user-event'
import { describe, it, expect, vi, beforeEach } from 'vitest'
import SIRForecast from '@/pages/forecasting/SIRForecast'
import { client } from '@/lib/api/client'

vi.mock('@/lib/api/client', () => ({
  client: {
    get: vi.fn(),
    post: vi.fn(),
  },
}))

// Mock Recharts
vi.mock('recharts', () => ({
  ResponsiveContainer: ({ children }: any) => <div data-testid="recharts-container">{children}</div>,
  LineChart: ({ children }: any) => <div data-testid="line-chart">{children}</div>,
  Line: () => <div data-testid="line" />,
  XAxis: () => <div />,
  YAxis: () => <div />,
  CartesianGrid: () => <div />,
  Tooltip: () => <div />,
  Legend: () => <div />,
}))

describe('SIR Forecast Component', () => {
  beforeEach(() => {
    vi.clearAllMocks()
    vi.mocked(client.get).mockResolvedValue({
      data: [
        { id: 1, name: 'Region 1' },
        { id: 2, name: 'Region 2' },
      ]
    })
    vi.mocked(client.post).mockResolvedValue({
      data: {
        forecast: [
          { day: 0, S: 1000, I: 10, R: 0 },
          { day: 1, S: 990, I: 20, R: 0 },
        ]
      }
    })
  })

  it('loads regions dynamically without hardcoded IDs', async () => {
    render(<SIRForecast />)
    
    // Check that getRegions API was called
    expect(client.get).toHaveBeenCalledWith('/regions/')

    // Should render the fetched regions
    await waitFor(() => {
      expect(screen.getByText('Region 1')).toBeInTheDocument()
      expect(screen.getByText('Region 2')).toBeInTheDocument()
    })
  })

  it('calls forecast API when region is selected', async () => {
    render(<SIRForecast />)
    
    await waitFor(() => {
      expect(screen.getByText('Region 1')).toBeInTheDocument()
    })

    // It should automatically request forecast for the first region (id: 1)
    await waitFor(() => {
      expect(client.post).toHaveBeenCalledWith('/models/sir/forecast', expect.objectContaining({
        region_id: 1,
        days: 30
      }))
    })
    
    // Change region
    const select = screen.getByRole('combobox')
    await userEvent.selectOptions(select, '2')
    
    // Should request for region 2
    expect(client.post).toHaveBeenCalledWith('/models/sir/forecast', expect.objectContaining({
      region_id: 2
    }))
    
    // Forecast chart should be rendered
    expect(screen.getByTestId('line-chart')).toBeInTheDocument()
  })
})
